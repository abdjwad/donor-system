⚡ الدوال الرئيسية

1. دالة التبرع — donate()

الدالة الأهم — تُستدعى عند كل تبرع بكريبتو Solidity 
/// @notice التبرع بـ MATIC لمشروع معين أو الصندوق العام
/// @param projectId معرف المشروع (0 = الصندوق العام)
/// @param message   رسالة إهداء اختيارية
function donate(
    uint256  projectId,
    string calldata message
) external payable nonReentrant {

    // ① التحقق من الشروط — إذا فشل، يُرجع الـ MATIC ويوقف التنفيذ
    require(msg.value >= MIN_DONATION, "amount below minimum");
    require(projectActive[projectId],  "project not active");

    // ② تسجيل التبرع بشكل دائم على الـ blockchain
    uint256 donationId = donations.length;
    donations.push(Donation({
        donor:     msg.sender,     // عنوان المتبرع تلقائياً
        amount:    msg.value,      // المبلغ المرسل بـ wei
        projectId: projectId,
        timestamp: block.timestamp, // وقت البلوكة
        message:   message
    }));

    // ③ تحديث الإحصائيات
    totalRaisedByProject[projectId] += msg.value;
    donationCountByDonor[msg.sender]++;
    totalRaised += msg.value;

    // ④ إطلاق الحدث — يصل للـ frontend فوراً عبر ethers.js
    emit DonationReceived(
        msg.sender, projectId, msg.value,
        donationId, block.timestamp, message
    );
}

2. دوال القراءة المجانية (View Functions)

الدالة ما ترجع الكلفة وين تُستخدم
getDonationsCount() عدد التبرعات الكلي مجاني إحصائيات الشفافية
getRecentDonations(count) آخر N تبرع مجاني صفحة الشفافية + الداشبورد

connectWallet — ربط MetaMask والاستماع لتغييراتها TypeScript 
async connectWallet(): Promise<void> {
    const eth = this.getEthereum(); // window.ethereum (MetaMask)

    // ① طلب الوصول للحسابات — MetaMask يفتح popup للمستخدم
    const accounts = await eth.request({ method: 'eth_requestAccounts' });

    if (accounts[0]) {
        this.walletAddress.set(accounts[0]);

        // ② قراءة الشبكة الحالية
        const hex = await eth.request({ method: 'eth_chainId' });
        this.chainId.set(parseInt(hex, 16));

        // ③ الاستماع لتغيير الحساب أو الشبكة
        eth.on('accountsChanged', (accs: string[]) =>
            this.walletAddress.set(accs[0] ?? null));
        eth.on('chainChanged', (hexId: string) =>
            this.chainId.set(parseInt(hexId, 16)));
    }
}
donate — إرسال التبرع للعقد الذكي TypeScript 
/**
 * إرسال تبرع للعقد الذكي
 * @returns TX Hash — رقم المعاملة على الـ blockchain للتحقق منها
 */
async donate(projectId: number, amountEth: string, message: string): Promise<string> {
    await this.loadAddresses();           // تحميل عنوان العقد من الـ JSON
    const contract = await this.getSignerContract(); // عقد يحتاج توقيع

    // MetaMask يفتح popup لتأكيد المعاملة
    const tx = await contract['donate'](projectId, message, {
        value: parseEther(amountEth), // تحويل "1.5" ETH → wei
    });

    // انتظار تأكيد البلوكة على الـ blockchain
    const receipt = await tx.wait();
    return receipt.hash; // TX Hash للتحقق
}

// الفرق بين Read Contract و Write Contract:

// للقراءة — يستخدم RPC مباشرة، مجاني، بدون MetaMask
private async getReadContract(): Promise<Contract> {
    const provider = await this.getProvider(); // BrowserProvider
    return new Contract(addr, DONATION_ABI, provider);
}

// للكتابة — يحتاج Signer (توقيع المستخدم في MetaMask)
private async getSignerContract(): Promise<Contract> {
    const provider = await this.getProvider();
    const signer   = await provider.getSigner(); // حساب المستخدم
    return new Contract(addr, DONATION_ABI, signer);
}

معالجة الأخطاء

ترجمة أخطاء ethers.js لرسائل واضحة TypeScript 
parseError(err: any): WalletError {
    if (err.code === 'NO_METAMASK')        return 'NO_METAMASK';
    if (err.code === 4001 ||
        err.code === 'ACTION_REJECTED')    return 'USER_REJECTED';
    if (err.code === 'INSUFFICIENT_FUNDS') return 'INSUFFICIENT_FUNDS';
    if (err.message?.includes('rejected')) return 'TX_REJECTED';
    return 'UNKNOWN';
}

// الـ WalletError type:
type WalletError =
    | 'NO_METAMASK'        // MetaMask غير مثبت
    | 'USER_REJECTED'      // المستخدم رفض في MetaMask
    | 'WRONG_NETWORK'      // شبكة خاطئة
    | 'INSUFFICIENT_FUNDS' // رصيد غير كافٍ
    | 'TX_REJECTED'        // المعاملة رُفضت من الـ blockchain
    | 'UNKNOWN';           // خطأ غير معروف

حالات الـ Transaction وكيف تتغير TypeScript 
type TxState = 'idle' | 'pending' | 'success' | 'error';

txState     = signal<TxState>('idle');    // حالة المعاملة
txHash      = signal<string | null>(null); // TX Hash بعد النجاح
txError     = signal<string | null>(null); // رسالة الخطأ
balance     = signal<string>('—');          // رصيد المحفظة
contractAddr = signal<string>('');          // عنوان العقد للعرض

async sendDonation(): Promise<void> {
    if (!this.web3.isConnected()) return;

    this.txState.set('pending');   // ① يُظهر spinner
    this.txError.set(null);

    try {
        // ② MetaMask يفتح — المستخدم يوافق
        const hash = await this.web3.donate(
            this.projectId, this.amount, this.message
        );
        this.txHash.set(hash);
        this.txState.set('success'); // ③ نجاح — يُظهر TX Hash
        this.balance.set(await this.web3.getWalletBalance()); // تحديث الرصيد
    } catch (err: any) {
        const code = this.web3.parseError(err);
        this.txError.set(this.errorMessage(code));
        this.txState.set('error');    // ④ فشل — يُظهر الخطأ
    }
}

خطوات التبرع خطوة بخطوة

1
المستخدم يملأ نموذج التبرع في CryptoPaymentComponent
المبلغ يأتي من DonationService، والرسالة من النموذج
2
sendDonation() تستدعي web3.donate(projectId, amount, message)
txState يصبح 'pending' → يظهر spinner
3
Web3Service يحمّل عنوان العقد من JSON ويبني Contract object
fetch('/assets/blockchain/contract-addresses.json')
ثم: new Contract(address, ABI, signer)
4
MetaMask يفتح popup للمستخدم — يعرض المبلغ + تكلفة الـ Gas
المستخدم يوافق أو يرفض — الرفض يُطلق خطأ ACTION_REJECTED
5
المعاملة تُرسَل للـ Ganache/Polygon وتنتظر التعدين
const receipt = await tx.wait()
على Ganache: فوري. على Polygon: ثوانٍ معدودة.
6
العقد يُنفِّذ donate() — يسجل التبرع ويُطلق DonationReceived
التبرع الآن دائم على الـ blockchain — لا يمكن لأحد محوه
7
receipt.hash يُعاد للـ component → txState = 'success'
يُعرض TX Hash للمستخدم — يستطيع نسخه والتحقق منه

🛡️ الأمان — كيف حمينا العقود

استخدمنا مكتبة OpenZeppelin الصناعية المعيارية لحماية الأموال.

🔁
ReentrancyGuard — منع هجمات إعادة الدخول
خطر: عقد خبيث يستدعي donate() مجدداً قبل انتهاء التنفيذ — يسرق الأموال. الحل: nonReentrant modifier يمنع أي استدعاء متداخل للدالة.
👑
Ownable — صلاحيات الإدارة
كل دوال الإدارة (سحب أموال، إيقاف مشروع، إنشاء مشروع) محمية بـ onlyOwner — فقط محفظة الإدارة تستطيع تنفيذها.
✅
Input Validation — التحقق من المدخلات
require(msg.value >= MIN_DONATION) — لا يقبل تبرع أقل من 0.1 MATIC (يمنع spam المعاملات).
require(projectActive[projectId]) — يرفض التبرع للمشاريع الموقوفة.
🔗
onlyDonationContract — ربط آمن بين العقدين
دالة updateAmountRaised() في BunianProject تقبل الاستدعاء فقط من عنوان BunianDonation — يمنع أي عقد آخر من تلاعب البيانات.
📱
Network Validation — التحقق من الشبكة في الفرونت
isWrongNetwork Signal تمنع إرسال أي معاملة إذا كان MetaMask على شبكة خاطئة — تحمي من إرسال MATIC للعقد على شبكة غلط.
🌐
isPlatformBrowser — SSR Safety
Web3Service يتحقق أنه يعمل في المتصفح قبل أي وصول لـ window.ethereum — يمنع الأعطال عند Server-Side Rendering.
⚠️
تحذير مهم للإنتاج: hardhat.config.js يحتوي على Private Keys مشفرة للاختبار فقط. قبل النشر على Polygon Mainnet، يجب استخدام متغيرات البيئة (.env) واستبدال المفاتيح بمفاتيح إنتاج آمنة.

── BLOCKCHAIN LAYER ──────────────────────────────────────────

BunianDonation.sol
    ← import OpenZeppelin/Ownable
    ← import OpenZeppelin/ReentrancyGuard

    → ganache.json يُنسَخ → src/assets/blockchain/contract-addresses.json

── ANGULAR LAYER ─────────────────────────────────────────────

web3.service.ts
    ← import ethers (BrowserProvider, Contract, parseEther, formatEther)
    ← import blockchain.model.ts (interfaces)
    ← يقرأ contract-addresses.json → يبني Contract objects
    → يُحقَن في: CryptoPaymentComponent, NavbarComponent

blockchain.model.ts
    → BlockchainDonation interface
    → ContractAddresses interface
    → WalletError type

crypto-payment.component.ts
    ← inject(Web3Service)

    → يستدعي: web3.connectWallet(), web3.donate(), web3.getWalletBalance()
    → يستدعي: web3.switchToGanache(), web3.parseError()

navbar.component.ts
    ← inject(Web3Service)
    → يعرض: walletAddress Signal, isConnected Signal
    → زر "Connect MetaMask" → web3.connectWallet()
